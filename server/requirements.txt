# Serveur MajiChrono — dependances de production.
#
# Versions epinglees au majeur : une montee de version doit etre un acte
# volontaire, verifie par la suite de tests, jamais une surprise de deploiement.
#
# En production (DirectAdmin/Passenger), l'application est servie via
# `passenger_wsgi.py` + a2wsgi : uvicorn n'est jamais importe par le
# processus. Ses extras "standard" (uvloop, httptools...) exigent une
# compilation native qui echoue souvent sur l'hebergement mutualise et sont
# inutiles ici. Pour lancer le serveur en local, voir requirements-dev.txt.

fastapi==0.115.6
a2wsgi==1.10.10
pydantic==2.10.4
pydantic-settings==2.7.0

# Base de donnees. PyMySQL est pur Python et fonctionne avec MySQL/MariaDB
# installe par XAMPP sans compilation native.
sqlalchemy==2.0.36
alembic==1.14.0
pymysql==1.1.1

# Le site web partage `users.password_hash` au format PHP bcrypt ($2y$).
bcrypt==4.2.1
pyjwt==2.10.1

# Client HTTP utilise par les integrations MajiPay et SMS (app/core/majipay.py,
# app/core/sms.py), et par la suite de tests. L'envoi d'e-mail passe par le
# module `smtplib` livre avec Python : aucun fournisseur n'est cable en dur,
# et changer d'expediteur ne coute que quatre variables d'environnement.
httpx==0.28.1

python-multipart==0.0.20
